# Adds TWO private components: the JS/TS code sandbox and its dependency bundler. Both are
# PRIVATE (service-chained), so neither is publicly routable: the sandbox runs untrusted
# code, and the bundler (experimental) fetches arbitrary npm from esm.sh. Keeping them
# separate isolates the "fetch npm" capability from the "run untrusted code" capability.
# Reach them from your own component through @spinframework/sandbox:
#
#   import { Interpreter } from "@spinframework/sandbox/interpreter";
#   const js = new Interpreter({
#     baseUrl:    "http://{{project-name | kebab_case}}.spin.internal",
#     bundlerUrl: "http://{{project-name | kebab_case}}-bundler.spin.internal",
#   });
#   await js.run(code, { lang: "javascript", packageJson: { dependencies: { /* ... */ } } });
#
# Add BOTH of these to your calling component's `allowed_outbound_hosts`:
#   "http://{{project-name | kebab_case}}.spin.internal"
#   "http://{{project-name | kebab_case}}-bundler.spin.internal"

# --- JS/TS sandbox: runs untrusted code (no esm.sh egress of its own) ---
[[trigger.http]]
route = { private = true }
component = "{{project-name | kebab_case}}"

[component.{{project-name | kebab_case}}]
source = { url = "https://github.com/fermyon/spin-agent-sandbox/releases/download/canary/ts-sandbox.wasm", digest = "sha256:5c815217be184d01ad9b1697273ae41dba22c8d1bb6fd47d831a2ebaf04fe30b" }
key_value_stores = ["default"]
# Outbound HTTP is denied until you list hosts here, e.g. ["https://api.example.com"]. Each
# request's `allowedHosts` can only narrow this list, and guard-io always refuses private and
# internal addresses.
allowed_outbound_hosts = []
[component.{{project-name | kebab_case}}.variables]
log_level = "info"

# --- Bundler (EXPERIMENTAL): fetches npm deps from esm.sh, runs no user code ---
# Uses the public https://esm.sh. To use a self-hosted mirror
# (https://github.com/esm-dev/esm.sh/blob/main/HOSTING.md), replace the host below and add
# `[component.<name>-bundler.variables] esm_base_url = "https://esm.example.com"`.
[[trigger.http]]
route = { private = true }
component = "{{project-name | kebab_case}}-bundler"

[component.{{project-name | kebab_case}}-bundler]
source = { url = "https://github.com/fermyon/spin-agent-sandbox/releases/download/canary/bundler.wasm", digest = "sha256:d9f9bfde7a3d24033937cb2af0fe346e0a8a1c47ccdd09f4bfd316ff49480033" }
allowed_outbound_hosts = ["https://esm.sh"]
